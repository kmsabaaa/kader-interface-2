import { auth, currentUser } from "@clerk/nextjs/server";
import { redirect } from "next/navigation";
import Link from "next/link";
import { Film, Calendar, CreditCard, Bookmark, Bell, Search, LayoutGrid, Camera, Wallet, ArrowRightLeft, Check, X, CheckCircle2, Briefcase, Users } from "lucide-react";
import { UserButton } from "@clerk/nextjs";
import { db } from "../../lib/db";
import NewProjectModal from "./NewProjectModal";
import NewListingModal from "./NewListingModal";
import CreatorProfileModal from "./CreatorProfileModal";
import NewServiceModal from "./NewServiceModal";
import { toggleUserRole, respondToBooking } from "./actions";

export default async function Dashboard({ searchParams }: { searchParams: Promise<{ tab?: string }> }) {
  const resolvedParams = await searchParams;
  const currentTab = resolvedParams?.tab || "overview";

  const { userId } = await auth();
  if (!userId) redirect("/");
  const user = await currentUser();

  const dbUser = await db.user.upsert({
    where: { clerkId: userId },
    update: {},
    create: { clerkId: userId },
  });

  const isProvider = dbUser.role === "PROVIDER";

  // FETCH DATA
  const projects = await db.project.findMany({
    where: { userId: dbUser.id },
    orderBy: { createdAt: "desc" },
  });

  const listings = await db.listing.findMany({
    where: { userId: dbUser.id },
    orderBy: { createdAt: "desc" },
  });

  const services = await db.service.findMany({
    where: { userId: dbUser.id },
    orderBy: { createdAt: "desc" },
  });

  const pendingRequests = await db.callSheetItem.findMany({
    where: {
      OR:[ { listing: { userId: dbUser.id } }, { service: { userId: dbUser.id } } ],
      status: "PENDING"
    },
    include: { project: true, listing: true, service: true },
    orderBy: { createdAt: "desc" }
  });

  const approvedRequests = await db.callSheetItem.findMany({
    where: {
      OR:[ { listing: { userId: dbUser.id } }, { service: { userId: dbUser.id } } ],
      status: "APPROVED"
    },
    include: { project: true, listing: true, service: true },
    orderBy: { startDate: "asc" }
  });

  // EARNINGS MATH
  let totalEarnings = 0;
  approvedRequests.forEach((req) => {
    let days = 1;
    if (req.startDate && req.endDate) {
      const diff = new Date(req.endDate).getTime() - new Date(req.startDate).getTime();
      days = Math.ceil(diff / (1000 * 3600 * 24));
      if (days <= 0) days = 1;
    }
    const itemPrice = req.listing?.pricePerDay || req.service?.pricePerDay || 0;
    totalEarnings += (itemPrice * days);
  });

  const getTabClass = (tabName: string) => {
    return currentTab === tabName
      ? "flex items-center gap-3 px-4 py-3 bg-white/5 text-amber-500 rounded-xl font-bold border border-white/10 transition-all"
      : "flex items-center gap-3 px-4 py-3 text-zinc-400 hover:text-white hover:bg-white/5 rounded-xl font-medium transition-colors";
  };

  return (
    <div className="min-h-screen bg-[#030303] text-white font-sans flex selection:bg-amber-500/30">
      
      {/* SIDEBAR - Fixed visibility */}
      <aside className="w-64 border-r border-white/5 bg-[#0a0a0a] flex flex-col shrink-0">
        <Link href="/" className="p-6 flex items-center gap-3 hover:opacity-80 transition-opacity">
          <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-amber-400 to-amber-600 flex items-center justify-center font-bold text-black text-xl">
            K
          </div>
          <span className="text-xl font-bold tracking-tight">Mission Control</span>
        </Link>
        
        <nav className="flex-1 px-4 flex flex-col gap-2 mt-4">
          <Link href="/dashboard?tab=overview" className={getTabClass("overview")}>
            <LayoutGrid className="w-5 h-5" /> Overview
          </Link>
          
          {!isProvider ? (
            <>
              <Link href="/dashboard?tab=projects" className={getTabClass("projects")}>
                <Film className="w-5 h-5" /> My Projects
              </Link>
              <Link href="/dashboard?tab=saved" className={getTabClass("saved")}>
                <Bookmark className="w-5 h-5" /> Saved Resources
              </Link>
            </>
          ) : (
            <>
              <Link href="/dashboard?tab=inventory" className={getTabClass("inventory")}>
                <Briefcase className="w-5 h-5" /> My Offerings
              </Link>
              <Link href="/dashboard?tab=calendar" className={getTabClass("calendar")}>
                <Calendar className="w-5 h-5" /> Booking Calendar
              </Link>
              <Link href="/dashboard?tab=earnings" className={getTabClass("earnings")}>
                <Wallet className="w-5 h-5" /> Earnings
              </Link>
            </>
          )}
          
          <Link href="/dashboard?tab=billing" className={getTabClass("billing")}>
            <CreditCard className="w-5 h-5" /> Billing & Vault
          </Link>
        </nav>

        <div className="p-4 mt-auto">
          <form action={toggleUserRole}>
            <button type="submit" className="w-full py-3 rounded-xl border border-white/10 text-sm font-bold text-zinc-300 hover:text-white hover:bg-white/5 transition-all flex items-center justify-center gap-2">
              <ArrowRightLeft className="w-4 h-4" />
              Switch to {isProvider ? "Consumer" : "Provider"}
            </button>
          </form>
        </div>
      </aside>

      {/* MAIN CONTENT */}
      <main className="flex-1 flex flex-col h-screen overflow-y-auto relative">
        <header className="h-20 border-b border-white/5 px-8 flex items-center justify-between shrink-0 bg-[#0a0a0a]/50 backdrop-blur-md sticky top-0 z-10">
          <div className="flex items-center bg-white/5 rounded-lg px-4 py-2 w-96 border border-white/5 focus-within:border-amber-500/50 transition-colors">
            <Search className="w-4 h-4 text-zinc-500 mr-2" />
            <input type="text" placeholder="Search dashboard..." className="bg-transparent text-sm w-full outline-none placeholder:text-zinc-600" />
          </div>
          <div className="flex items-center gap-6">
            <button className="relative text-zinc-400 hover:text-amber-400 transition-colors">
              <Bell className="w-5 h-5" />
              {pendingRequests.length > 0 && <span className="absolute -top-1 -right-1 w-2.5 h-2.5 bg-amber-500 rounded-full border-2 border-[#0a0a0a]"></span>}
            </button>
            <div className="pl-6 border-l border-white/10">
              <UserButton appearance={{ elements: { avatarBox: "w-10 h-10 border-2 border-amber-500/50 hover:scale-105 transition-all" } }} />
            </div>
          </div>
        </header>

        <div className="p-8 max-w-6xl mx-auto w-full">
          <div className="flex flex-col md:flex-row justify-between md:items-end gap-6 mb-10">
            <div>
              <h1 className="text-3xl font-bold mb-2 capitalize">
                {currentTab === "overview" ? (isProvider ? "Provider Dashboard." : `Welcome back, ${user?.firstName || "Director"}.`) : currentTab === "inventory" ? "My Offerings" : currentTab.replace("-", " ")}
              </h1>
              <p className="text-zinc-400">
                {currentTab === "overview" 
                  ? (isProvider ? "Manage your equipment, locations, and talent bookings." : "Here is the status of your active productions.")
                  : `Manage your ${currentTab === "inventory" ? "offerings" : currentTab.replace("-", " ")}.`}
              </p>
            </div>
            
            {/* ACTION BUTTONS */}
            {!isProvider ? (
              <NewProjectModal />
            ) : (
              <div className="flex flex-wrap gap-4">
                <CreatorProfileModal user={dbUser} />
                {dbUser.isCreator && <NewServiceModal />}
                <NewListingModal />
              </div>
            )}
          </div>

          {/* TAB: OVERVIEW */}
          {currentTab === "overview" && (
            <>
              {/* METRICS */}
              <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mb-10">
                <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-6 relative overflow-hidden">
                  <div className="absolute top-0 right-0 w-32 h-32 bg-amber-500/10 blur-[50px]"></div>
                  <h3 className="text-zinc-400 text-sm font-medium mb-2">
                    {isProvider ? "Active Assets & Services" : "Active Productions"}
                  </h3>
                  <p className="text-4xl font-bold text-white">{isProvider ? (listings.length + services.length) : projects.length}</p> 
                </div>
                <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-6 relative overflow-hidden">
                  <div className="absolute top-0 right-0 w-32 h-32 bg-emerald-500/10 blur-[50px]"></div>
                  <h3 className="text-zinc-400 text-sm font-medium mb-2">
                    {isProvider ? "Total Earnings" : "Escrow Vault Balance"}
                  </h3>
                  <p className="text-4xl font-bold text-emerald-400">
                    {isProvider ? totalEarnings.toFixed(2) : "0"} <span className="text-xl text-zinc-500">BHD</span>
                  </p>
                </div>
                <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-6 relative overflow-hidden">
                  <div className="absolute top-0 right-0 w-32 h-32 bg-blue-500/10 blur-[50px]"></div>
                  <h3 className="text-zinc-400 text-sm font-medium mb-2">
                    {isProvider ? "Pending Requests" : "Pending AI Matches"}
                  </h3>
                  <p className="text-4xl font-bold text-white">{isProvider ? pendingRequests.length : "0"}</p>
                </div>
              </div>

              {/* INBOX NOTIFICATIONS */}
              {isProvider && pendingRequests.length > 0 && (
                <div className="bg-amber-500/5 border border-amber-500/20 rounded-2xl p-6 shadow-[0_0_30px_rgba(245,158,11,0.05)] mb-10">
                  <h3 className="text-lg font-bold text-amber-500 mb-4 flex items-center gap-2"><Bell className="w-5 h-5" /> Action Required: Booking Requests</h3>
                  <div className="space-y-4">
                    {pendingRequests.map((req) => {
                      const startDateStr = req.startDate ? new Date(req.startDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) : 'TBD';
                      const endDateStr = req.endDate ? new Date(req.endDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) : 'TBD';
                      let rentalDays = 1;
                      if (req.startDate && req.endDate) {
                        const diff = new Date(req.endDate).getTime() - new Date(req.startDate).getTime();
                        rentalDays = Math.ceil(diff / (1000 * 3600 * 24));
                        if (rentalDays <= 0) rentalDays = 1;
                      }
                      const itemName = req.listing?.title || req.service?.title || "Resource";
                      const itemPrice = req.listing?.pricePerDay || req.service?.pricePerDay || 0;
                      const totalEarn = itemPrice * rentalDays;

                      return (
                        <div key={req.id} className="bg-black/50 border border-white/10 rounded-xl p-5 flex flex-col md:flex-row justify-between items-start md:items-center gap-4">
                          <div>
                            <p className="text-white font-bold text-lg">{itemName}</p>
                            <p className="text-zinc-400 text-sm mb-2">Requested for project: <span className="text-white font-medium">{req.project.title}</span></p>
                            <div className="flex items-center gap-2 text-xs font-bold text-zinc-400 bg-white/5 border border-white/10 w-fit px-3 py-1.5 rounded-lg mb-2">
                              <Calendar className="w-3.5 h-3.5 text-amber-500" />
                              <span>{startDateStr} — {endDateStr} ({rentalDays} {rentalDays === 1 ? 'day' : 'days'})</span>
                            </div>
                            <p className="text-emerald-400 text-sm font-bold mt-1">Payout: {totalEarn.toFixed(2)} BHD</p>
                          </div>
                          <div className="flex items-center gap-3 w-full md:w-auto">
                            <form action={respondToBooking.bind(null, req.id, "DECLINED")} className="flex-1 md:flex-none">
                              <button type="submit" className="w-full bg-red-500/10 hover:bg-red-500/20 text-red-500 border border-red-500/30 font-bold rounded-lg px-4 py-2 flex items-center justify-center gap-2 transition-colors"><X className="w-4 h-4" /> Decline</button>
                            </form>
                            <form action={respondToBooking.bind(null, req.id, "APPROVED")} className="flex-1 md:flex-none">
                              <button type="submit" className="w-full bg-emerald-500 hover:bg-emerald-400 text-black font-bold rounded-lg px-6 py-2 flex items-center justify-center gap-2 transition-colors"><Check className="w-4 h-4" /> Approve</button>
                            </form>
                          </div>
                        </div>
                      )
                    })}
                  </div>
                </div>
              )}

              {/* RECENT PREVIEWS */}
              <h2 className="text-xl font-bold mb-6 flex items-center gap-2 border-t border-white/10 pt-10">
                {isProvider ? <Briefcase className="w-5 h-5 text-blue-500" /> : <Film className="w-5 h-5 text-amber-500" />} 
                {isProvider ? "Recent Offerings" : "Recent Projects"}
              </h2>

              {!isProvider && projects.length === 0 && <p className="text-zinc-500">No projects yet.</p>}
              {!isProvider && projects.slice(0, 2).map((project) => (
                <Link href={`/project/${project.id}`} key={project.id} className="block mb-4">
                  <div className="bg-zinc-900/50 border border-white/5 rounded-xl p-6 hover:border-amber-500/30 transition-colors">
                    <h3 className="text-xl font-bold text-white mb-1">{project.title}</h3>
                    <div className="flex justify-between items-center text-sm font-medium pt-2">
                      <span className="px-3 py-1 bg-amber-500/10 text-amber-500 rounded-full border border-amber-500/20">{project.status}</span>
                    </div>
                  </div>
                </Link>
              ))}

              {isProvider && (
                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  {services.slice(0, 2).map((service) => (
                    <div key={service.id} className="bg-zinc-900/50 border border-white/5 rounded-xl p-4 flex items-center gap-4">
                      <div className="w-12 h-12 bg-amber-500/10 text-amber-500 rounded-lg flex items-center justify-center shrink-0 border border-amber-500/20">
                        <Users className="w-6 h-6" />
                      </div>
                      <div>
                        <h3 className="text-sm font-bold text-white mb-0.5 line-clamp-1">{service.title}</h3>
                        <span className="text-amber-400 font-bold text-xs">{service.pricePerDay} BHD / day</span>
                      </div>
                    </div>
                  ))}
                  {listings.slice(0, 2).map((listing) => (
                    <div key={listing.id} className="bg-zinc-900/50 border border-white/5 rounded-xl p-4 flex items-center gap-4">
                      <div className="w-12 h-12 bg-zinc-800 rounded-lg overflow-hidden shrink-0 border border-white/10">
                        {listing.imageUrl ? <img src={listing.imageUrl} className="w-full h-full object-cover" alt={listing.title} /> : <Camera className="w-5 h-5 m-auto mt-3 text-zinc-500"/>}
                      </div>
                      <div>
                        <h3 className="text-sm font-bold text-white mb-0.5 line-clamp-1">{listing.title}</h3>
                        <span className="text-blue-400 font-bold text-xs">{listing.pricePerDay} BHD / day</span>
                      </div>
                    </div>
                  ))}
                  {services.length === 0 && listings.length === 0 && (
                     <p className="text-zinc-500 md:col-span-2">Your inventory is empty.</p>
                  )}
                </div>
              )}
            </>
          )}

          {/* TAB: PROJECTS (CONSUMER) */}
          {currentTab === "projects" && !isProvider && (
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {projects.length === 0 ? <p className="text-zinc-500">No projects found.</p> : null}
              {projects.map((project) => (
                <Link href={`/project/${project.id}`} key={project.id} className="block h-full">
                  <div className="bg-zinc-900/50 border border-white/5 rounded-xl p-6 hover:border-amber-500/30 transition-colors h-full flex flex-col">
                    <h3 className="text-xl font-bold text-white mb-1">{project.title}</h3>
                    <p className="text-zinc-400 text-sm mb-4 flex-grow">{project.description}</p>
                    <div className="flex justify-between items-center text-sm font-medium mt-auto pt-4 border-t border-white/5">
                      <span className="px-3 py-1 bg-amber-500/10 text-amber-500 rounded-full border border-amber-500/20">{project.status}</span>
                      <span className="text-zinc-300">Budget: {project.budget} BHD</span>
                    </div>
                  </div>
                </Link>
              ))}
            </div>
          )}

          {/* TAB: INVENTORY / OFFERINGS (PROVIDER) */}
          {currentTab === "inventory" && isProvider && (
            <div className="space-y-12">
              {/* Human Services */}
              {(dbUser.isCreator || services.length > 0) && (
                <div>
                  <h2 className="text-xl font-bold mb-6 flex items-center gap-2"><Briefcase className="w-5 h-5 text-amber-500" /> Your Service Roster</h2>
                  {services.length === 0 ? (
                    <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-8 flex flex-col items-center text-center border-dashed">
                      <p className="text-zinc-400">No services listed. Click Add Service above.</p>
                    </div>
                  ) : (
                    <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                      {services.map((service) => (
                        <div key={service.id} className="bg-amber-500/5 border border-amber-500/20 rounded-xl p-6">
                          <h3 className="text-xl font-bold text-white mb-1">{service.title}</h3>
                          <p className="text-zinc-400 text-sm mb-4">{service.description}</p>
                          <div className="flex justify-between items-center text-sm font-medium pt-4 border-t border-white/5">
                            <span className="text-amber-500 font-bold">{service.pricePerDay} BHD / day</span>
                          </div>
                        </div>
                      ))}
                    </div>
                  )}
                </div>
              )}

              {/* Physical Assets */}
              <div>
                <h2 className="text-xl font-bold mb-6 flex items-center gap-2"><Camera className="w-5 h-5 text-blue-500" /> Your Asset Inventory</h2>
                {listings.length === 0 ? (
                  <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-12 flex flex-col items-center justify-center text-center border-dashed">
                    <p className="text-zinc-400">No assets listed.</p>
                  </div>
                ) : (
                  <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                    {listings.map((listing) => (
                      <div key={listing.id} className="group bg-zinc-900/50 border border-white/5 rounded-2xl overflow-hidden hover:border-blue-500/30 transition-colors">
                        <div className="h-48 w-full bg-zinc-800 relative overflow-hidden">
                          {listing.imageUrl && <img src={listing.imageUrl} alt={listing.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500" />}
                          <div className="absolute top-3 left-3 px-2 py-1 bg-black/60 backdrop-blur-md rounded-md border border-white/10 text-xs font-bold text-white uppercase tracking-wider">{listing.type}</div>
                        </div>
                        <div className="p-5">
                          <h3 className="text-lg font-bold text-white mb-1">{listing.title}</h3>
                          <p className="text-zinc-400 text-sm mb-4 line-clamp-2">{listing.description}</p>
                          <div className="flex justify-between items-center pt-4 border-t border-white/5">
                            <span className="text-blue-400 font-bold">{listing.pricePerDay} BHD <span className="text-zinc-500 text-xs font-normal">/ day</span></span>
                          </div>
                        </div>
                      </div>
                    ))}
                  </div>
                )}
              </div>
            </div>
          )}

          {/* TAB: CALENDAR (PROVIDER) */}
          {currentTab === "calendar" && isProvider && (
            <div>
              <h2 className="text-xl font-bold mb-6 flex items-center gap-2"><CheckCircle2 className="w-5 h-5 text-emerald-500" /> Confirmed Upcoming Bookings</h2>
              {approvedRequests.length === 0 ? (
                <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-12 flex flex-col items-center text-center border-dashed">
                  <p className="text-zinc-400">Your schedule is completely clear.</p>
                </div>
              ) : (
                <div className="grid grid-cols-1 gap-4">
                  {approvedRequests.map((req) => {
                    const startDateStr = req.startDate ? new Date(req.startDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }) : '';
                    const endDateStr = req.endDate ? new Date(req.endDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }) : '';
                    
                    const isHuman = !!req.service;
                    const itemName = req.listing?.title || req.service?.title || "Resource";
                    const itemImage = req.listing?.imageUrl || null;

                    return (
                      <div key={req.id} className="bg-black/50 border border-white/5 hover:border-emerald-500/30 transition-colors rounded-xl p-4 flex flex-col md:flex-row justify-between items-center gap-4">
                        <div className="flex items-center gap-4">
                          <div className={`w-12 h-12 rounded-lg overflow-hidden shrink-0 border border-white/10 flex items-center justify-center ${isHuman ? 'bg-amber-500/10 text-amber-500' : 'bg-zinc-800 text-zinc-600'}`}>
                            {itemImage ? <img src={itemImage} className="w-full h-full object-cover" alt="gear" /> : (isHuman ? <Briefcase className="w-6 h-6" /> : <Camera className="w-6 h-6" />)}
                          </div>
                          <div>
                            <p className="text-white font-bold">{itemName}</p>
                            <p className="text-zinc-500 text-sm">
                              {isHuman ? "Hired for: " : "Rented to: "} 
                              <span className="text-zinc-300">{req.project.title}</span>
                            </p>
                          </div>
                        </div>
                        <div className="text-right">
                          <p className="text-emerald-400 font-bold text-sm">
                            {isHuman ? "Call Time / Gig Start" : "Prepare For Hand-off"}
                          </p>
                          <p className="text-zinc-400 text-xs font-mono mt-0.5">{startDateStr} ➔ {endDateStr}</p>
                        </div>
                      </div>
                    )
                  })}
                </div>
              )}
            </div>
          )}

          {/* TAB: EARNINGS / BILLING / SAVED */}
          {(currentTab === "earnings" || currentTab === "billing" || currentTab === "saved") && (
            <div className="bg-zinc-900/50 border border-white/5 rounded-2xl p-16 flex flex-col items-center justify-center text-center border-dashed">
              <h3 className="text-2xl font-bold text-white mb-2">Module Offline</h3>
              <p className="text-zinc-400 max-w-md mb-6">This module requires Stripe Escrow integration. We will activate it in the next architecture phase.</p>
            </div>
          )}

        </div>
      </main>
    </div>
  );
}
